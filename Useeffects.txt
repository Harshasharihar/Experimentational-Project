useEffect(() => {
  if (!benchmark) return;

  const loadCapacity = async () => {
    try {
      const response = await fetch(
        `${GLOBAL_BACKEND_URL}/api/benchmarks/${encodeURIComponent(
          benchmark
        )}/filters/capacity`,
        {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
          },
          body: JSON.stringify({
            // add your Swagger payload here
          }),
        }
      );

      const data = await response.json();

      setCapacityOptions(
        Array.isArray(data)
          ? data
          : data?.capacity ||
              data?.capacities ||
              data?.data ||
              []
      );
    } catch (error) {
      console.error("Error loading capacity:", error);
      setCapacityOptions([]);
    }
  };

  loadCapacity();
}, [benchmark]);

useEffect(() => {
  if (!benchmark || selectedScenarios.length < 2) {
    setTableData(null);
    setGraphData([]);
    return;
  }

  const loadScenarioResults = async () => {
    try {
      setLoadingResults(true);

      const response = await fetch(
        `${GLOBAL_BACKEND_URL}/api/benchmarks/${encodeURIComponent(
          benchmark
        )}/sc-results`,
        {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
          },
          body: JSON.stringify({
            // add your Swagger payload here
          }),
        }
      );

      const data = await response.json();

      setTableData(data?.table || null);
      setGraphData(
        Array.isArray(data?.graph)
          ? data.graph
          : data?.graphs || []
      );
    } catch (error) {
      console.error("Error loading scenario results:", error);
      setTableData(null);
      setGraphData([]);
    } finally {
      setLoadingResults(false);
    }
  };

  loadScenarioResults();
}, [benchmark, selectedScenarios]);
